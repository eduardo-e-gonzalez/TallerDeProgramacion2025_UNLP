programa E3P2
procesos
  proceso juntarPapeles(ES papeles:numero)
  comenzar
    mientras (HayPapelEnLaEsquina)
      tomarPapel
      papeles:=papeles+1
  fin
  proceso juntarFlores (ES flores:numero)
  comenzar
    mientras (HayFlorEnLaEsquina)
      tomarFlor
      flores:=flores+1
  fin
areas
  a1:AreaP(1,1,10,1)
  a2:AreaP(11,1,20,1)
  b1:AreaP(1,5,10,5)
  b2:AreaP(11,5,20,5)
robots
  robot rob1
  variables
    papeles:numero
  comenzar
    papeles:=0
    derecha
    repetir 9
      juntarPapeles(papeles)
      mover
    juntarPapeles(papeles)
    EnviarMensaje(papeles,A2)
  fin
  robot rob2
  variables
    papeles:numero
    papeles1:numero
  comenzar
    papeles:=0
    derecha
    RecibirMensaje(papeles1,A1)
    repetir 9
      juntarPapeles(papeles)
      mover
    juntarPapeles(papeles)
    papeles:=papeles+papeles1
    Informar(papeles)
  fin
  robot rob3
  variables
    flores:numero
  comenzar
    flores:=0
    derecha
    repetir 9
      juntarFlores(flores)
      mover
    juntarFlores(flores)
    EnviarMensaje (flores,B2)
  fin
  robot rob4
  variables
    flores:numero
    flores1:numero
  comenzar
    flores:=0
    derecha
    RecibirMensaje(flores1,B1)
    repetir 9
      juntarFlores(flores)
      mover
    juntarFlores(flores)
    flores:=flores+flores1
    Informar(flores)
  fin
variables
  A1:rob1
  A2:rob2
  B1:rob3
  B2:rob4
comenzar
  AsignarArea(A1,a1)
  AsignarArea(A2,a2)
  AsignarArea(B1,b1)
  AsignarArea(B2,b2)
  Iniciar(A1,1,1)
  Iniciar(A2,11,1)
  Iniciar(B1,1,5)
  Iniciar(B2,11,5)
fin
