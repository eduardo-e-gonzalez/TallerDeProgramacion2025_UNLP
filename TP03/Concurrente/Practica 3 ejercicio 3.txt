programa E3P3
procesos
  proceso recogerFlor (ES flor:numero;E av:numero;E ca:numero;ES seguir:boolean)
  variables
    avAct,caAct:numero
  comenzar
    avAct:=PosAv
    caAct:=PosCa
    BloquearEsquina (av,ca)
    Pos (av,ca)
    si (HayFlorEnLaEsquina)
      tomarFlor
      flor:=flor+1
    sino
      seguir:=F
    Pos(avAct,caAct)
    LiberarEsquina(av,ca)
  fin
  proceso Enviar
  variables
    av,ca:numero
  comenzar
    Random(av,2,10)
    Random(ca,2,10)
    EnviarMensaje(1,r1)
    EnviarMensaje(2,r2)
    EnviarMensaje(3,r3)
    EnviarMensaje(4,r4)
      
    EnviarMensaje(av,r1)
    EnviarMensaje(ca,r1)
      
    EnviarMensaje(av,r2)
    EnviarMensaje(ca,r2)
      
    EnviarMensaje(av,r3)
    EnviarMensaje(ca,r3)
      
    EnviarMensaje(av,r4)
    EnviarMensaje(ca,r4)
  fin
areas
  a1:AreaP(2,1,2,1)
  a2:AreaP(3,1,3,1)
  a3:AreaP(4,1,4,1)
  a4:AreaP(5,1,5,1)
  aJ:AreaP(1,1,1,1)
  cuadrante: AreaPC (2,2,10,10)
robots
  robot rob
  variables
    ID,av,ca:numero
    flor:numero
    seguir:boolean
  comenzar
    flor:=0
    seguir:=V
    RecibirMensaje(ID,j)
    RecibirMensaje(av,j)
    RecibirMensaje(ca,j)
    mientras (seguir)
      recogerFlor(flor,av,ca,seguir)
    EnviarMensaje(ID,j)
    EnviarMensaje(flor,j)
  fin
  robot jefe
  variables
    ganador:numero
    may,flores:numero
    ID:numero
  comenzar
    Enviar
    may:=-1
    repetir 4
      RecibirMensaje (ID,*)
      si (ID=1)
        RecibirMensaje (flores,r1)
      sino
        si (ID=2)
          RecibirMensaje (flores,r2)
        sino
          si (ID=3)
            RecibirMensaje (flores,r3)
          sino
            si (ID=4)
              RecibirMensaje (flores,r4)
      si (flores>may)
        may:=flores
        ganador:=ID
    Informar("El ganador es ",ganador)
    Informar("Con ",may)
  fin
variables
  r1,r2,r3,r4:rob
  j:jefe
comenzar
  AsignarArea(j,aJ)
  AsignarArea(r1,a1)
  AsignarArea(r1,cuadrante)
  AsignarArea(r2,a2)
  AsignarArea(r2,cuadrante)
  AsignarArea(r3,a3)
  AsignarArea(r3,cuadrante)
  AsignarArea(r4,a4)
  AsignarArea(r4,cuadrante)
  Iniciar (j,1,1)
  Iniciar(r1,2,1)
  Iniciar (r2,3,1)
  Iniciar(r3,4,1)
  Iniciar (r4,5,1)
fin
